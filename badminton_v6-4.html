CREATE TABLE IF NOT EXISTS pending_matches (
  id bigserial PRIMARY KEY,
  type text,
  team_a jsonb,
  team_b jsonb,
  score_a int,
  score_b int,
  win_team text,
  pts_gain int,
  pts_loss int,
  submitted_by bigint,
  created_at timestamptz DEFAULT now()
);
ALTER TABLE pending_matches ENABLE ROW LEVEL SECURITY;
CREATE POLICY "allow all" ON pending_matches FOR ALL USING (true) WITH CHECK (true);

🏅 ระบบ Rank & เงื่อนไข
คะแนน = True ELO Rating
ชนะคนแรงค์สูง → ได้เยอะ | แพ้คนแรงค์ต่ำ → เสียเยอะ
K-Factor: ผู้เล่นใหม่ K=48 · ระดับกลาง K=32 · เก่า K=24
Score Multiplier: สูสี ×1.0 · ขาด ×1.5 · ขาดลอย ×2.0

✅ มีได้แค่ 1 คนเท่านั้นในระบบ
📌 สูตร True ELO
• E = 1 / (1 + 10^((คู่แข่ง - ผู้เล่น) / 400))
• Gain = K × (1 - E) × ScoreMult
• Loss = K × E × ScoreMult
• King: ต้องมีคะแนนสูงสุด AND ≥ 2,000 คะแนน
• คะแนนขั้นต่ำ 0 · ได้/เสียขั้นต่ำ 4 ต่อแมตช์
